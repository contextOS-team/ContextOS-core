   ```ts

   import { createClient } from '@/lib/supabase/client'
   import { getDashboardAttention, fetchMyWorkGraph, type AttentionItem } from '@/lib/api'
   import { fetchContextPages, type ContextPageRow } from '@/lib/context-feed'
   import type { LensGraphResponse } from '@/components/graph/types'
   import { fetchActivityDelta, bucketDelta } from './since-last-visit/activity-delta'
   import type { Delta, NeedsYouItem } from './types'

   export interface DashboardData {
     decisions: LensGraphResponse | null
     contextItems: ContextPageRow[]
     needsYouItems: NeedsYouItem[]
     delta: Delta
     workspaceId: string | null
     isLoading: boolean
     error: string | null

   export function useDashboardData(lastVisitedAt: Date | null): DashboardData {
     // Local state, parallel Promise.allSettled loading mirroring existing page.tsx pattern
     // (lines 85-93). Adapt the result into the new shape:
     //  - decisions = fetchMyWorkGraph result (used by RecentDecisions component to filter entity_type=='Decision')
     //  - contextItems = top 3 from fetchContextPages
     //  - needsYouItems = mapped from getDashboardAttention(); each AttentionItem becomes a NeedsYouItem

     //         endpoint is per-user)

     //      - person: the asker / source person
     //      - sinceDay: humanize via a small helper — Mon/Tue/Wed if within 7 days, else "MMM d"
     //      - ageMs: now - item.createdAt
     //  - activityRows = fetchActivityDelta(workspaceId, lastVisitedAt); then delta = bucketDelta(activityRows, needsYouItems, currentUserId)
     // Return everything in one object.
   }
   ```
   - **Crucial:** the hook accepts `lastVisitedAt` as a param (passed down from `page.tsx`) so it can re-run `fetchActivityDelta` if/when `lastVisitedAt` changes. import { RecentDecisions } from './pulse/recent-decisions'
```tsx
import { ActiveContext } from './pulse/active-context'
import { NeedsYou } from './pulse/needs-you'
import { useLastVisitedAt } from './pulse/since-last-visit/use-last-visited-at'
import { SeenItemsProvider } from './pulse/since-last-visit/use-session-seen-items'
import { useDashboardData } from './pulse/use-dashboard-data'

export default function DashboardPage() {
  // Resolve workspaceId first (needed by useLastVisitedAt and useDashboardData)
  const [workspaceId, setWorkspaceId] = useState<string | null>(null)
  useEffect(() => {
    let cancelled = false
    const supabase = createClient()
    supabase.auth.getUser().then(async (res) => {
      const u = res.data?.user
      if (!u) return
      const { data: m } = await supabase
        .from('workspace_members')
        .select('workspace_id')
        .eq('user_id', u.id)
        .single()
      if (!cancelled && m?.workspace_id) setWorkspaceId(m.workspace_id)
    })
    return () => { cancelled = true }
  }, [])

  // Stable per-session value
  const lastVisitedAt = useLastVisitedAt(workspaceId)

  // Wait for both workspaceId AND the initial lastVisitedAt read before loading data.
  // (null lastVisitedAt is valid — it just means first-visit; but null workspaceId means we
  // haven't resolved the user's workspace yet, which is a transient loading state.)
  return (
    <SeenItemsProvider>
      <main className="pulse-surface min-h-screen">
        <div className="mx-auto max-w-5xl px-12 py-16 space-y-24">
          <DashboardSections workspaceId={workspaceId} lastVisitedAt={lastVisitedAt} />
        </div>
      </main>
    </SeenItemsProvider>
  )
}

function DashboardSections({ workspaceId, lastVisitedAt }: { workspaceId: string | null; lastVisitedAt: Date | null }) {
  const { decisions, contextItems, needsYouItems, delta, isLoading, error } = useDashboardData(lastVisitedAt)

  if (isLoading) {
    return (
      <>
        <Skeleton className="h-12 w-96" />
        <Skeleton className="h-4 w-64 mt-6" />
        <div className="mt-24">
          <Skeleton className="h-4 w-40 mb-6" />
          <div className="space-y-6">
            <Skeleton className="h-32 w-full" />
            <Skeleton className="h-32 w-full" />
          </div>
        </div>
      </>
    )
  }

  if (error) {
    return <p className="text-sm" style={{ color: '#b06863' }}>Failed to load dashboard: {error}</p>
  }

  return (
    <>
      <Headline delta={delta} lastVisitedAt={lastVisitedAt} />
      <RecentDecisions data={decisions} lastVisitedAt={lastVisitedAt} />
      <ActiveContext items={contextItems} lastVisitedAt={lastVisitedAt} />
      <NeedsYou items={needsYouItems} lastVisitedAt={lastVisitedAt} />
    </>
  )
}
```
