```typescript
// Settings - API Keys
export interface ApiKeyResponse {
  id: string
  name: string
  key_prefix: string
  created_at: string
}

export interface ApiKeyCreatedResponse extends ApiKeyResponse {
  key: string  // Full key, shown once only
}

export function listApiKeys(): Promise<{ keys: ApiKeyResponse[] }>
export function createApiKey(name: string): Promise<ApiKeyCreatedResponse>
export function revokeApiKey(keyId: string): Promise<{ status: string }>

// Settings - Workspace
export function updateWorkspaceName(name: string): Promise<{ id: string; name: string; slug: string; created_at: string }>

// Settings - Team
export interface TeamMember {
  user_id: string
  full_name: string | null
  email: string
  role: string
  created_at: string
}

export function listTeamMembers(): Promise<{ members: TeamMember[] }>
```

Implement each function using `apiFetch<T>()` with the correct HTTP method and path:

- `updateWorkspaceName` -> PATCH `/api/workspaces/current` with `{ name }` body
- `listTeamMembers` -> GET `/api/workspaces/current/members`
