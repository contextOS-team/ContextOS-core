    Create `apps/web/src/app/dashboard/integrations/slack-dm-toggle.tsx`:

    ```tsx
    'use client'

    import { useEffect, useState } from 'react'
    import { Switch } from '@/components/ui/switch'
    import { Label } from '@/components/ui/label'
    import { toast } from 'sonner'
    import { getSlackStatus, toggleSlackDms } from '@/lib/api'

    interface Props {
      // Optional — when provided, parent can avoid an extra fetch.
      initialIncludeDms?: boolean
    }

    export function SlackDmToggle({ initialIncludeDms }: Props) {
      const [enabled, setEnabled] = useState<boolean>(initialIncludeDms ?? false)
      const [loading, setLoading] = useState<boolean>(initialIncludeDms === undefined)
      const [pending, setPending] = useState(false)

      useEffect(() => {
        if (initialIncludeDms !== undefined) return
        let cancelled = false
        ;(async () => {
          try {
            const s = await getSlackStatus()
            if (!cancelled) setEnabled(!!s.include_dms)
          } catch {
            // Non-blocking — leave default false
          } finally {
            if (!cancelled) setLoading(false)
          }
        })()
        return () => { cancelled = true }
      }, [initialIncludeDms])

      async function handleChange(next: boolean) {
        // Optimistic update
        const prev = enabled
        setEnabled(next)
        setPending(true)
        try {
          const res = await toggleSlackDms(next)
          setEnabled(res.include_dms)
          toast.success(res.include_dms ? 'DM ingestion enabled' : 'DM ingestion disabled')
        } catch (err) {
          setEnabled(prev)
          toast.error(err instanceof Error ? err.message : 'Failed to update DM setting')
        } finally {
          setPending(false)
        }
      }

      if (loading) {
        // Minimal skeleton — matches .planning conventions (no flash of default state)
        return <div className="h-10 w-full animate-pulse rounded-md bg-muted/40" aria-hidden="true" />
      }

      return (
        <div className="space-y-1.5 rounded-md border border-border bg-background p-3">
          <div className="flex items-center justify-between gap-4">
            <Label htmlFor="slack-dm-toggle" className="flex-1 cursor-pointer">
              <span className="text-xs font-medium text-foreground">Include direct messages</span>
            </Label>
            <Switch
              id="slack-dm-toggle"
              checked={enabled}
              onCheckedChange={handleChange}
              disabled={pending}
              // Ensure >= 40px hit area via a wrapper label/Switch; verify Switch default is 44×24 — if not, add an outer pseudo-element on the label
            />
          </div>
          <p
            className="text-[11px] text-muted-foreground"
            style={{ textWrap: 'pretty' }}
          >
            DM messages will be indexed for context generation. Only you can see this data.
          </p>
        </div>
      )
    }
    ```
