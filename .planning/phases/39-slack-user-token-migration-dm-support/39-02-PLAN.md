    ```python
    async def poll_user_messages(session: AsyncSession, workspace_id: UUID) -> dict:
        integration = await _get_integration(session, workspace_id)
        if integration is None or integration.status != "connected":
            return {"ingested": 0, "skipped": 0, "reason": "not_connected"}
        config = integration.config or {}

        if not config.get("include_dms"):
            return {"ingested": 0, "skipped": 0, "reason": "dms_disabled"}

        try:
            user_token = await get_user_token(session, workspace_id)
        except SlackOAuthError:
            logger.warning("Slack user token missing for ws %s — needs_reauth", workspace_id)
            return {"ingested": 0, "skipped": 0, "reason": "needs_reauth"}

        is_backfill = bool(config.get("dms_first_enabled_at")) and not config.get("dms_backfill_done")
        oldest = _compute_oldest(config, is_backfill)

        ingested = skipped = 0
        needs_reauth_hit = False

        async with httpx.AsyncClient(timeout=30.0) as client:
            try:
                dm_ids = await _enumerate_dms(client, user_token)
            except _NeedsReauth:
                return {"ingested": 0, "skipped": 0, "reason": "needs_reauth"}

            for conv_id in dm_ids:
                try:
                    messages = await _fetch_history(
                        client, user_token, conv_id, oldest,
                        cap=(BACKFILL_MESSAGES_PER_CONVERSATION_CAP if is_backfill else None),
                    )
                except _NeedsReauth:
                    needs_reauth_hit = True
                    break
                # Task 2 wires the real upsert here — for Task 1, stub to (0, 0).
                ins, skp = await _upsert_messages(session, workspace_id, conv_id, messages)
                ingested += ins
                skipped += skp

        if needs_reauth_hit:
            return {"ingested": ingested, "skipped": skipped, "reason": "needs_reauth"}

        # Update config (preserve other keys)
        updated = dict(integration.config) if integration.config else {}
        updated["last_user_poll_at"] = datetime.now(timezone.utc).replace(tzinfo=None).isoformat()
        if is_backfill:
            updated["dms_backfill_done"] = True
        integration.config = updated
        await session.commit()

        return {"ingested": ingested, "skipped": skipped, "reason": None}
    ```

    ```python
    async def _upsert_messages(
        session: AsyncSession,
        workspace_id: UUID,
        conv_id: str,
        messages: list[dict],
    ) -> tuple[int, int]:
        if not messages:
            return (0, 0)

        # Filter first — mirrors slack_events._handle_message contract
        filtered: list[dict] = []
        skipped_filter = 0
        for msg in messages:
            if _should_filter(msg):
                skipped_filter += 1
                continue
            filtered.append(msg)

        if not filtered:
            return (0, skipped_filter)

        # Pre-count existing rows for dedup bookkeeping (Test J)
        ts_list = [m["ts"] for m in filtered]
        existing_stmt = select(SlackMessage.slack_ts).where(
            SlackMessage.workspace_id == workspace_id,
            SlackMessage.channel_id == conv_id,
            SlackMessage.slack_ts.in_(ts_list),
        )
        existing = {row[0] for row in (await session.execute(existing_stmt)).all()}

        inserted_ids: list[UUID] = []
        for msg in filtered:
            values = {
                "workspace_id": workspace_id,
                "slack_ts": msg["ts"],
                "channel_id": conv_id,
                "channel_name": None,
                "user_id": msg.get("user"),
                "user_name": None,
                "text": msg.get("text", ""),
                "thread_ts": msg.get("thread_ts"),
                "parent_ts": (
                    msg.get("thread_ts")
                    if msg.get("thread_ts") and msg["thread_ts"] != msg["ts"]
                    else None
                ),
                "is_thread_reply": bool(
                    msg.get("thread_ts") and msg["thread_ts"] != msg["ts"]
                ),
                "slack_created_at": _ts_to_datetime(msg["ts"]),
                "raw_data": msg,
            }
            stmt = (
                pg_insert(SlackMessage)
                .values(**values)
                .on_conflict_do_update(
                    constraint="uq_slack_msg",
                    # SET clause parity with slack_events._handle_message — see SLACK-USR-04.
                    # Fields MUST match: text, user_id, thread_ts, parent_ts, is_thread_reply, raw_data.
                    set_={
                        k: pg_insert(SlackMessage).excluded[k]
                        for k in (
                            "text",
                            "user_id",
                            "thread_ts",
                            "parent_ts",
                            "is_thread_reply",
                            "raw_data",
                        )
                    },
                )
                .returning(SlackMessage.id)
            )
            result = await session.execute(stmt)
            row_id = result.scalar_one()
            inserted_ids.append(row_id)

        await session.commit()

        # Embedding + graph extraction parity with slack_events._handle_message.
        for msg, row_id in zip(filtered, inserted_ids, strict=True):
            await embedding.queue_embedding_generation(
                session, workspace_id, "slack_message", [row_id]
            )
            try:
                await extract_and_populate_graph(
                    session=session,
                    workspace_id=workspace_id,
                    content=msg.get("text", ""),
                    source_type="slack_message",
                    source_id=row_id,
                    source_name=f"Slack DM in {conv_id}",
                    source_external_id=msg["ts"],
                )
            except Exception:
                logger.warning(
                    "graph extraction failed for slack_message %s (non-fatal)",
                    row_id,
                    exc_info=True,
                )

        # ingested = newly inserted rows (ts not in existing)
        # skipped = filter-skips + dedup-updates (ts was in existing)
        ingested = sum(1 for m in filtered if m["ts"] not in existing)
        skipped = skipped_filter + sum(1 for m in filtered if m["ts"] in existing)
        return (ingested, skipped)
    ```
