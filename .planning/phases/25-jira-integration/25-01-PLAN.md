   - priority (text) -- Jira priority is string, not int

   - assignee_name (text)
   - reporter_name (text)

   - project_name (text)
   - labels (text[])

   - updated_at (timestamptz NOT NULL DEFAULT now())
   - deleted_at (timestamptz)
   - UNIQUE (workspace_id, jira_id)

2. **jira_comments table:**
   - id (uuid PK, default gen_random_uuid())
   - workspace_id (uuid FK -> workspaces, NOT NULL, ON DELETE CASCADE)
   - issue_id (uuid FK -> jira_issues, NOT NULL, ON DELETE CASCADE)
   - jira_id (text NOT NULL) -- Jira comment ID
   - author_name (text)
   - body (text)
   - jira_created_at (timestamptz)
   - jira_updated_at (timestamptz)
   - raw_data (jsonb NOT NULL DEFAULT '{}')
