2. Add a new `find_source_content()` function to `linking.py` that finds content for a SINGLE source type. This function does NOT change the existing `find_related_content()` -- it wraps the same query patterns but filtered per source.

The approach (per RESEARCH.md recommendation): Run the existing `_find_explicit_refs` and `_find_hybrid_matches` once, then group/filter results by source type. This avoids per-source DB queries while still yielding per-source progress.

Add a new function:
```python
SOURCE_TYPE_MAP = {
    "slack": ["slack_message"],
    "github": ["github_pull_request", "github_review", "code_chunk"],
    "notion": ["notion_page"],
    "fathom": ["meeting_transcript", "meeting_segment"],
    "linear": ["linear_comment"],
}

async def find_content_by_source(
    session: AsyncSession,
    workspace_id: UUID,
    ticket: LinearTicket,
) -> AsyncGenerator[tuple[str, list[LinkedItem], str], None]:
    """Find related content and yield results grouped by source.

    Yields (source_name, items, detail_string) tuples for each source
    that has content. Runs all queries upfront, then groups by source.
    """
    # Run existing queries
    explicit = await _find_explicit_refs(session, workspace_id, ticket)
    hybrid = await _find_hybrid_matches(session, workspace_id, ticket)

    # Merge
    merged: dict[tuple[str, UUID], LinkedItem] = {}
    for signal_items in [explicit, hybrid]:
        for key, item in signal_items.items():
            if key in merged:
                merged[key].relevance_score += item.relevance_score
                merged[key].signals.extend(item.signals)
            else:
                merged[key] = item

    # Group by source
    for source_name, source_types in SOURCE_TYPE_MAP.items():
        source_items = [
            item for item in merged.values()
            if item.source_type in source_types
        ]
        source_items.sort(key=lambda x: x.relevance_score, reverse=True)

        # Build detail string
        if source_name == "github":
            prs = sum(1 for i in source_items if i.source_type == "github_pull_request")
            reviews = sum(1 for i in source_items if i.source_type == "github_review")
            code = sum(1 for i in source_items if i.source_type == "code_chunk")
            parts = []
            if prs: parts.append(f"{prs} PR{'s' if prs > 1 else ''}")
            if reviews: parts.append(f"{reviews} review{'s' if reviews > 1 else ''}")
            if code: parts.append(f"{code} code file{'s' if code > 1 else ''}")
            detail = f"Found {', '.join(parts)}" if parts else "No PRs or code found"
        elif source_name == "slack":
            n = len(source_items)
            detail = f"Found {n} message{'s' if n != 1 else ''}" if n else "No messages found"
        elif source_name == "notion":
            n = len(source_items)
            detail = f"Found {n} page{'s' if n != 1 else ''}" if n else "No pages found"
        elif source_name == "fathom":
            n = len(source_items)
            detail = f"Found {n} meeting segment{'s' if n != 1 else ''}" if n else "No meetings found"
        elif source_name == "linear":
            n = len(source_items)
            detail = f"Found {n} comment{'s' if n != 1 else ''}" if n else "No comments found"
        else:
            detail = f"Found {len(source_items)} items" if source_items else "No items found"

        yield source_name, source_items, detail
```

2. In `context.py`, add the SSE endpoint:

```python
from sse_starlette import EventSourceResponse, ServerSentEvent
import json

@router.get("/{ticket_identifier}/generate-stream")
async def stream_context_generation(
    ticket_identifier: str,
    user: UserContext = Depends(get_current_user),
    settings: Settings = Depends(get_settings),
):
    """Stream context generation progress via SSE.

    CRITICAL: Create a dedicated session INSIDE the generator, not via Depends.
    FastAPI closes Depends sessions when the response object is returned,
    which for SSE is immediately -- before the stream ends.
    """
    async def event_generator():
        async with async_session() as session:
            try:
                # Check for concurrent generation
                stmt = select(ContextPage).where(
                    ContextPage.workspace_id == user.workspace_id,
                    ContextPage.ticket_identifier == ticket_identifier,
                    ContextPage.status == "generating",
                    ContextPage.deleted_at.is_(None),
                )
                result = await session.execute(stmt)
                if result.scalar_one_or_none():
                    yield ServerSentEvent(
                        data=json.dumps({"event": "error", "message": "Generation already in progress"}),
                        event="error",
                    )
                    return

                async for event in generate_context_page_stream(
                    session, user.workspace_id, ticket_identifier
                ):
                    yield ServerSentEvent(
                        data=json.dumps(event),
                        event=event["event"],
                    )

                await session.commit()
            except Exception as e:
                logger.exception("SSE generation failed for %s", ticket_identifier)
                await session.rollback()
                yield ServerSentEvent(
                    data=json.dumps({"event": "error", "message": str(e)}),
                    event="error",
                )

    return EventSourceResponse(event_generator())
```
