    ```python
    async def claim_chat(session, message: dict) -> ClaimResult:
        chat = message.get("chat", {})
        chat_type = chat.get("type")
        if chat_type not in ("group", "supergroup"):
            return ClaimResult.IGNORE_PRIVATE

        text = message.get("text", "") or ""
        parts = text.split(maxsplit=1)
        code = parts[1].strip() if len(parts) >= 2 else ""

        chat_id = chat["id"]
        chat_title = chat.get("title")

        # Step 1: peek the code (no delete) to know which workspace it belongs to
        code_row = await session.execute(
            select(TelegramPairingCode.workspace_id, TelegramPairingCode.created_by)
            .where(
                TelegramPairingCode.code == code,
                TelegramPairingCode.expires_at > func.now(),
            )
        )
        code_record = code_row.first()
        existing_ws = await resolve_workspace_for_chat(session, chat_id)

        if code_record is None:
            # No valid code. Reply with invalid/expired regardless of mapping state.
            await _send_reply_safe(chat_id, _reply_text(ClaimResult.INVALID_CODE))
            return ClaimResult.INVALID_CODE

        claimer_ws, created_by = code_record

        # Cross-workspace claim attempt — REJECT, do NOT burn code (someone else may use it
        # legitimately later; codes are scoped to a workspace).
        if existing_ws is not None and existing_ws != claimer_ws:
            await _send_reply_safe(chat_id, _reply_text(ClaimResult.ALREADY_CLAIMED_BY_OTHER))
            return ClaimResult.ALREADY_CLAIMED_BY_OTHER

        # Same-workspace re-claim — NOOP, don't burn the code (CONTEXT idempotency rule).
        if existing_ws is not None and existing_ws == claimer_ws:
            return ClaimResult.NOOP

        # Fresh claim — atomically burn the code and insert the mapping.
        redeem_stmt = (
            delete(TelegramPairingCode)
            .where(
                TelegramPairingCode.code == code,
                TelegramPairingCode.expires_at > func.now(),
            )
            .returning(TelegramPairingCode.workspace_id)
        )
        burned = (await session.execute(redeem_stmt)).first()
        if burned is None:
            # Race: someone else burned it between peek and delete. Treat as invalid.
            await _send_reply_safe(chat_id, _reply_text(ClaimResult.INVALID_CODE))
            return ClaimResult.INVALID_CODE

        ins_stmt = pg_insert(TelegramChatWorkspace).values(
            chat_id=chat_id,
            workspace_id=claimer_ws,
            chat_title=chat_title,
            chat_type=chat_type,
            paired_by=created_by,
        ).on_conflict_do_nothing(index_elements=["chat_id"])
        ins_result = await session.execute(ins_stmt)
        inserted = ins_result.rowcount == 1

        await emit_activity(
            session, claimer_ws, "integration_connected",
            metadata={"integration": "telegram", "chat_title": chat_title, "chat_id": chat_id},
            actor_id=created_by,
        )
        await session.commit()

        if inserted:
            await _send_reply_safe(chat_id, _reply_text(ClaimResult.CLAIMED, chat_title=chat_title))
            return ClaimResult.CLAIMED
        # Race: chat became mapped between our resolve_workspace_for_chat and INSERT.
        # The code was already burned. Best-effort — treat as ALREADY_CLAIMED.
        await _send_reply_safe(chat_id, _reply_text(ClaimResult.ALREADY_CLAIMED_BY_OTHER))
        return ClaimResult.ALREADY_CLAIMED_BY_OTHER

        if not settings.TELEGRAM_BOT_TOKEN_GLOBAL:
            logger.warning("telegram_reply_skipped chat_id=%s reason=no_global_token", chat_id)
            return
        try:
            await telegram_api_call(
                settings.TELEGRAM_BOT_TOKEN_GLOBAL,
                "sendMessage",
                chat_id=chat_id,
                text=text,
                disable_web_page_preview=True,
            )
        except (TelegramError, Exception):
            logger.exception("telegram_reply_failed chat_id=%s", chat_id)

    def _reply_text(result: ClaimResult, *, chat_title: str | None = None) -> str:
        # Wording is Claude's discretion per CONTEXT. Keep short and link-driven.
        dashboard_url = "https://contextos.app/dashboard/settings"  # TODO replace with config-derived URL if available
        if result == ClaimResult.CLAIMED:
            return f"ContextOS connected{f' to {chat_title}' if chat_title else ''}. Messages here will be indexed for context generation. Manage at {dashboard_url}."
        if result == ClaimResult.INVALID_CODE:
            return f"This pairing link expired or was already used. Generate a new one at {dashboard_url}."
        if result == ClaimResult.ALREADY_CLAIMED_BY_OTHER:
            return "This chat is already paired with another ContextOS workspace."
        return ""

    async def handle_my_chat_member(session, update: dict) -> None:
        chat_id = update.get("chat", {}).get("id")
        new_status = update.get("new_chat_member", {}).get("status")
        logger.info("telegram_my_chat_member chat_id=%s new_status=%s", chat_id, new_status)
        if new_status not in ("left", "kicked"):
            return
        # Look up workspace_id for activity event before soft-delete
        ws_id = await resolve_workspace_for_chat(session, chat_id)
        await session.execute(
            sa_update(TelegramChatWorkspace)
            .where(TelegramChatWorkspace.chat_id == chat_id)
            .values(removed_at=func.now())
        )
        await session.commit()
        if ws_id is not None:
            await emit_activity(
                session, ws_id, "integration_disconnected",
                metadata={"integration": "telegram", "chat_id": chat_id, "reason": new_status},
            )
            await session.commit()
    ```
