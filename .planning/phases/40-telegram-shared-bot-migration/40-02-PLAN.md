       ```typescript
       'use client'

       import { useCallback, useEffect, useRef, useState } from 'react'
       import { Loader2, ExternalLink, RefreshCw, X, AlertTriangle, Check } from 'lucide-react'
       import { toast } from 'sonner'
       import { Button } from '@/components/ui/button'
       import { createClient } from '@/lib/supabase/client'
       import { createTelegramPairingCode } from '@/lib/api'
       import type { RealtimePostgresChangesPayload } from '@supabase/supabase-js'

       type Phase = 'idle' | 'minting' | 'waiting' | 'expired' | 'connected'

       interface TelegramPairingProgressProps {
         /** Current authenticated user's workspace UUID — used for Realtime filter */
         workspaceId: string
         /** Called when a telegram_chat_workspace row INSERT is observed for this workspace */
         onConnected?: (chat: { chat_id: number; chat_title: string | null }) => void
         /** Called when the user clicks Cancel or after a successful claim — parent should refresh statuses */
         onClose?: () => void
       }

       export function TelegramPairingProgress({ workspaceId, onConnected, onClose }: TelegramPairingProgressProps) {
         const [phase, setPhase] = useState<Phase>('idle')
         const [code, setCode] = useState<string | null>(null)
         const [deepLink, setDeepLink] = useState<string | null>(null)
         const [expiresAt, setExpiresAt] = useState<Date | null>(null)
         const [secondsLeft, setSecondsLeft] = useState<number>(0)

         // Use refs for callbacks to avoid stale closures on re-subscription
         const onConnectedRef = useRef(onConnected)
         onConnectedRef.current = onConnected

         // Mint a new pairing code (also used by Regenerate)
         const startPairing = useCallback(async () => {
           setPhase('minting')
           try {
             const result = await createTelegramPairingCode()
             setCode(result.code)
             setDeepLink(result.deep_link)
             setExpiresAt(new Date(result.expires_at))
             setPhase('waiting')
             // Open the deep link in a new tab so the user lands in the Telegram app/web
             window.open(result.deep_link, '_blank', 'noopener,noreferrer')
           } catch (err) {
             toast.error(err instanceof Error ? err.message : 'Failed to start pairing')
             setPhase('idle')
           }
         }, [])

         // Countdown ticker — updates secondsLeft once per second; flips to 'expired' at 0
         useEffect(() => {
           if (phase !== 'waiting' || !expiresAt) return
           const tick = () => {
             const remaining = Math.max(0, Math.floor((expiresAt.getTime() - Date.now()) / 1000))
             setSecondsLeft(remaining)
             if (remaining === 0) {
               setPhase('expired')
             }
           }
           tick()
           const id = setInterval(tick, 1000)
           return () => clearInterval(id)
         }, [phase, expiresAt])

         // Realtime subscription on telegram_chat_workspace inserts for this workspace
         useEffect(() => {
           if (phase !== 'waiting' || !workspaceId) return
           const supabase = createClient()
           const channel = supabase
             .channel(`telegram-pairing-${workspaceId}`)
             .on(
               'postgres_changes',
               {
                 event: 'INSERT',
                 schema: 'public',
                 table: 'telegram_chat_workspace',
                 filter: `workspace_id=eq.${workspaceId}`,
               },
               (payload: RealtimePostgresChangesPayload<Record<string, unknown>>) => {
                 const row = payload.new as { chat_id?: number; chat_title?: string | null }
                 if (typeof row.chat_id === 'number') {
                   setPhase('connected')
                   onConnectedRef.current?.({ chat_id: row.chat_id, chat_title: row.chat_title ?? null })
                 }
               }
             )
             .subscribe()
           return () => {
             supabase.removeChannel(channel)
           }
         }, [phase, workspaceId])

         // Auto-start the flow when the component mounts (IntegrationCard renders connectForm
         // only after the user clicks Connect — so mounting === user intent).
         useEffect(() => {
           if (phase === 'idle') {
             startPairing()
           }
           // eslint-disable-next-line react-hooks/exhaustive-deps
         }, [])

         function handleCancel() {
           setPhase('idle')
           onClose?.()
         }

         async function handleRegenerate() {
           await startPairing()
         }

         // RENDER ---------------------------------------------------------------
         if (phase === 'minting') {
           return (
             <div className="flex items-center gap-2 text-xs text-muted-foreground">
               <Loader2 className="h-3.5 w-3.5 animate-spin" />
               Generating pairing link…
             </div>
           )
         }

         if (phase === 'connected') {
           return (
             <div className="flex items-center gap-2 text-xs text-emerald-500">
               <Check className="h-3.5 w-3.5" />
               Connected — refreshing…
             </div>
           )
         }

         if (phase === 'expired') {
           return (
             <div className="space-y-3">
               <div className="flex items-center gap-2 text-xs text-amber-500">
                 <AlertTriangle className="h-3.5 w-3.5" />
                 This pairing link expired.
               </div>
               <div className="flex items-center gap-2">
                 <Button size="sm" onClick={handleRegenerate}>
                   <RefreshCw className="mr-1.5 h-3.5 w-3.5" />
                   Regenerate
                 </Button>
                 <Button size="sm" variant="ghost" onClick={handleCancel}>
                   Cancel
                 </Button>
               </div>
             </div>
           )
         }

         // phase === 'waiting'
         const minutes = Math.floor(secondsLeft / 60)
         const seconds = secondsLeft % 60
         const countdown = `${minutes}:${seconds.toString().padStart(2, '0')}`

         return (
           <div className="space-y-3">
             <p className="text-xs text-muted-foreground">
               Waiting for @ContextOSBot to join a Telegram group… (Choose a group when the
               Telegram app opens; the bot will post a confirmation in-chat.)
             </p>
             {deepLink && (
               <a
                 href={deepLink}
                 target="_blank"
                 rel="noopener noreferrer"
                 className="inline-flex items-center gap-1.5 text-xs text-primary hover:underline"
               >
                 <ExternalLink className="h-3.5 w-3.5" />
                 Reopen Telegram link
               </a>
             )}
             <div className="flex items-center gap-2">
               <Button size="sm" variant="ghost" onClick={handleCancel}>
                 <X className="mr-1.5 h-3.5 w-3.5" />
                 Cancel
               </Button>
               <Button size="sm" variant="outline" onClick={handleRegenerate}>
                 <RefreshCw className="mr-1.5 h-3.5 w-3.5" />
                 Regenerate
               </Button>
               <span className="ml-auto font-mono text-[11px] text-muted-foreground/70">
                 {countdown}
               </span>
             </div>
           </div>
         )
       }
       ```

       ```typescript
       'use client'

       import { useEffect, useState } from 'react'
       import { TelegramPairingProgress } from '../../integrations/telegram-pairing-progress'
       import { createClient } from '@/lib/supabase/client'

       export function TelegramConnect({ onConnected }: { onConnected?: () => void }) {
         const [workspaceId, setWorkspaceId] = useState<string | null>(null)

         // Resolve workspace_id from the authenticated session.
         // NOTE for executor: Look in apps/web/src/app/dashboard for an existing hook that
         // returns workspace_id (e.g., useUser, useSession, useWorkspace). If found, replace
         // this block with that hook. The fallback below queries Supabase directly.
         useEffect(() => {
           let cancelled = false
           const run = async () => {
             const supabase = createClient()
             const { data: { user } } = await supabase.auth.getUser()
             if (!user || cancelled) return
             // Workspace lookup — match the existing pattern used by SlackDmToggle /
             // GithubRepos. If those use a memoized `getWorkspaceId()` helper, use it here.
             const { data: ws } = await supabase
               .from('workspace_members')
               .select('workspace_id')
               .eq('user_id', user.id)
               .limit(1)
               .maybeSingle()
             if (!cancelled && ws?.workspace_id) {
               setWorkspaceId(ws.workspace_id as string)
             }
           }
           run()
           return () => { cancelled = true }
         }, [])

         if (!workspaceId) {
           return <p className="text-xs text-muted-foreground">Loading…</p>
         }

         return (
           <TelegramPairingProgress
             workspaceId={workspaceId}
             onConnected={() => onConnected?.()}
             onClose={onConnected}
           />
         )
       }
       ```
