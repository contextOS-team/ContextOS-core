    1. Create `backend/scripts/purge_legacy_telegram_vault.py`. Outline:

       ```python
       """One-shot script to purge legacy per-workspace Telegram bot tokens from Vault.

       Phase 40 cutover: Phase-26 stored each workspace's bot token in Vault under the key
       `telegram_<workspace_id>_bot_token` (via vault.token_name). Plan 01's migration deletes
       the corresponding oauth_tokens rows, but Vault entries are not regular DB rows — they
       must be cleaned up via the Vault API.

       Run AFTER deploying the Plan 01 migration (so the oauth_tokens rows are already gone)
       and AFTER the new global webhook is registered (so no Phase-26 connect attempts fire).

       Usage:
           cd backend && uv run python scripts/purge_legacy_telegram_vault.py [--dry-run]
       """
       import argparse
       import asyncio
       import logging
       import re

       from sqlalchemy import text as sa_text
       from src.db import async_session
       from src.services.vault import delete_token

       logger = logging.getLogger(__name__)
       NAME_PATTERN = re.compile(r"^telegram_[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}_bot_token$")


       async def list_legacy_telegram_secrets(session) -> list[str]:
           """Return Vault secret names matching the Phase-26 telegram pattern."""
           # Vault secrets live in vault.secrets (Supabase pgvault extension).
           # Filter SQL-side to avoid pulling unrelated secret names into Python.
           result = await session.execute(
               sa_text("SELECT name FROM vault.secrets WHERE name LIKE 'telegram_%_bot_token'")
           )
           return [row[0] for row in result.all() if NAME_PATTERN.match(row[0])]


       async def main(dry_run: bool) -> None:
           async with async_session() as session:
               names = await list_legacy_telegram_secrets(session)
               logger.info("Found %d legacy Telegram Vault entries", len(names))
               if dry_run:
                   for name in names:
                       logger.info("[DRY-RUN] would delete: %s", name)
                   return
               for name in names:
                   try:
                       await delete_token(session, name)
                       logger.info("deleted: %s", name)
                   except Exception:
                       logger.exception("failed to delete %s — continuing", name)
               await session.commit()


       if __name__ == "__main__":
           logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
           parser = argparse.ArgumentParser()
           parser.add_argument("--dry-run", action="store_true", help="List entries without deleting")
           args = parser.parse_args()
           asyncio.run(main(args.dry_run))
       ```
