**3. Create `supabase/migrations/00002_workspace_creation_trigger.sql`:**

Database trigger on `auth.users` INSERT that atomically:
1. Creates a workspace (name = user's email prefix, slug = sanitized email prefix + random suffix)
2. Creates a profile row in `public.profiles`
3. Creates a workspace_members row linking user to workspace with role 'owner'

All in one transaction. Per user decision: "Supabase DB trigger on auth.users INSERT creates workspace + user profile atomically."

```sql
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
DECLARE
  new_workspace_id uuid;
  email_prefix text;
  workspace_slug text;
BEGIN
  -- Extract email prefix for workspace naming
  email_prefix := split_part(NEW.email, '@', 1);
  workspace_slug := lower(regexp_replace(email_prefix, '[^a-z0-9]', '-', 'g')) || '-' || substr(gen_random_uuid()::text, 1, 8);

  -- Create workspace
  INSERT INTO public.workspaces (name, slug)
  VALUES (email_prefix || '''s Workspace', workspace_slug)
  RETURNING id INTO new_workspace_id;

  -- Create profile
  INSERT INTO public.profiles (id, email, full_name)
  VALUES (NEW.id, NEW.email, COALESCE(NEW.raw_user_meta_data->>'full_name', email_prefix));

  -- Create membership
  INSERT INTO public.workspace_members (workspace_id, user_id, role)
  VALUES (new_workspace_id, NEW.id, 'owner');

  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();
```
